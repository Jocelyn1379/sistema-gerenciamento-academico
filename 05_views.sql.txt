-- Seleciona o banco de dados utilizado no projeto.
USE sistema_academico;

-- =====================================================
-- CRIAÇÃO DA VIEW DE RELATÓRIO DE MATRÍCULAS
-- A VIEW reúne informações de alunos, disciplinas,
-- professores, turmas e matrículas em uma única consulta.
-- =====================================================

CREATE OR REPLACE VIEW vw_relatorio_matriculas AS
SELECT
    a.matricula,
    a.nome AS aluno,
    d.nome AS disciplina,
    p.nome AS professor,
    t.codigo_turma,
    t.ano,
    t.semestre,
    m.data_matricula,
    m.situacao
FROM matricula m

-- Relaciona as matrículas aos respectivos alunos.
INNER JOIN aluno a
    ON m.matricula_aluno = a.matricula

-- Relaciona as matrículas às turmas correspondentes.
INNER JOIN turma t
    ON m.codigo_turma = t.codigo_turma

-- Obtém o nome da disciplina de cada turma.
INNER JOIN disciplina d
    ON t.codigo_disc = d.codigo_disc

-- Obtém o nome do professor responsável pela turma.
INNER JOIN professor p
    ON t.codigo_prof = p.codigo_prof;

-- =====================================================
-- CONSULTA GERAL DA VIEW
-- Exibe todos os registros disponíveis no relatório.
-- =====================================================

SELECT * FROM vw_relatorio_matriculas;

-- =====================================================
-- CONSULTA DE MATRÍCULAS ATIVAS
-- Utiliza WHERE para exibir somente as matrículas
-- cuja situação está definida como "Ativa".
-- =====================================================

SELECT *
FROM vw_relatorio_matriculas
WHERE situacao = 'Ativa';
